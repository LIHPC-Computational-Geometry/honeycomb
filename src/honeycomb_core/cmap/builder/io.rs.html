23pub(crate) fn parse_meta(meta_line: &str) -> Result<(String, usize, usize), BuilderError> {
24    let parts: Vec<&str> = meta_line.split_whitespace().collect();
25    if parts.len() != 3 {
26        return Err(BuilderError::BadMetaData("incorrect format"));
27    }
28
29    Ok((
30        parts[0].to_string(),
31        parts[1]
32            .parse()
33            .map_err(|_| BuilderError::BadMetaData("could not parse dimension"))?,
34        parts[2]
35            .parse()
36            .map_err(|_| BuilderError::BadMetaData("could not parse dart number"))?,
37    ))
38}

40impl TryFrom<String> for CMapFile {
41    type Error = BuilderError;
42
43    fn try_from(value: String) -> Result<Self, Self::Error> {
44        let mut sections = HashMap::default();
45        let mut current_section = String::new();
46
47        for line in value.trim().lines() {
48            let trimmed = line.trim();
49            if trimmed.is_empty() || trimmed.starts_with('#') {
50                // ignore empty & comment lines
51                continue;
52            }
53            if trimmed.starts_with('[') && trimmed.contains(']') {
54                // process section header
55                let section_name = trimmed.trim_matches(['[', ']']).to_lowercase();
56
57                if section_name != "meta"
58                    && section_name != "betas"
59                    && section_name != "unused"
60                    && section_name != "vertices"
61                {
62                    return Err(BuilderError::UnknownHeader(section_name));
63                }
64
65                if sections
66                    .insert(section_name.clone(), String::new())
67                    .is_some()
68                {
69                    return Err(BuilderError::DuplicatedSection(section_name));
70                }
71                current_section = section_name;
72
73                continue;
74            }
75            if !current_section.is_empty() {
76                // regular line
77                let line_without_comment = trimmed.split('#').next().unwrap().trim();
78                if !line_without_comment.is_empty() {
79                    let current_content = sections.get_mut(&current_section).unwrap();
80                    if !current_content.is_empty() {
81                        current_content.push('\n');
82                    }
83                    current_content.push_str(line_without_comment);
84                }
85            }
86        }
87
88        if !sections.contains_key("meta") {
89            // missing required section
90            return Err(BuilderError::MissingSection("meta"));
91        }
92        if !sections.contains_key("betas") {
93            // missing required section
94            return Err(BuilderError::MissingSection("betas"));
95        }
96
97        Ok(Self {
98            meta: parse_meta(sections["meta"].as_str())?,
99            betas: sections["betas"].clone(),
100            unused: sections.get("unused").cloned(),
101            vertices: sections.get("vertices").cloned(),
102        })
103    }
104}

108pub fn build_2d_from_cmap_file<T: CoordsFloat>(
109    f: CMapFile,
110    manager: AttrStorageManager, // FIXME: find a cleaner solution to populate the manager
111) -> Result<CMap2<T>, BuilderError> {
112    if f.meta.1 != 2 {
113        // mismatched dim
114        return Err(BuilderError::BadMetaData(
115            "mismatch between requested dimension and header",
116        ));
117    }
118    let map = CMap2::new_with_undefined_attributes(f.meta.2, manager);
119
120    // putting it in a scope to drop the data
121    let betas = f.betas.lines().collect::<Vec<_>>();
122    if betas.len() != 3 {
123        // mismatched dim
124        return Err(BuilderError::InconsistentData(
125            "wrong number of beta functions",
126        ));
127    }
128    let b0 = betas[0]
129        .split_whitespace()
130        .map(str::parse)
131        .collect::<Vec<_>>();
132    let b1 = betas[1]
133        .split_whitespace()
134        .map(str::parse)
135        .collect::<Vec<_>>();
136    let b2 = betas[2]
137        .split_whitespace()
138        .map(str::parse)
139        .collect::<Vec<_>>();
140
141    // mismatched dart number
142    if b0.len() != f.meta.2 + 1 {
143        return Err(BuilderError::InconsistentData(
144            "wrong number of values for the beta 0 function",
145        ));
146    }
147    if b1.len() != f.meta.2 + 1 {
148        return Err(BuilderError::InconsistentData(
149            "wrong number of values for the beta 1 function",
150        ));
151    }
152    if b2.len() != f.meta.2 + 1 {
153        return Err(BuilderError::InconsistentData(
154            "wrong number of values for the beta 2 function",
155        ));
156    }
157
158    for (d, b0d, b1d, b2d) in multizip((
159        (1..=f.meta.2),
160        b0.into_iter().skip(1),
161        b1.into_iter().skip(1),
162        b2.into_iter().skip(1),
163    )) {
164        let b0d = b0d.map_err(|_| BuilderError::BadValue("could not parse a b0 value"))?;
165        let b1d = b1d.map_err(|_| BuilderError::BadValue("could not parse a b1 value"))?;
166        let b2d = b2d.map_err(|_| BuilderError::BadValue("could not parse a b2 value"))?;
167        map.set_betas(d as DartIdType, [b0d, b1d, b2d]);
168    }
169
170    if let Some(unused) = f.unused {
171        for u in unused.split_whitespace() {
172            let d = u
173                .parse()
174                .map_err(|_| BuilderError::BadValue("could not parse an unused ID"))?;
175            map.release_dart(d)
176                .expect("E: unused dart has non-null beta images");
177        }
178    }
179
180    if let Some(vertices) = f.vertices {
181        for l in vertices.trim().lines() {
182            let mut it = l.split_whitespace();
183            let id: VertexIdType = it
184                .next()
185                .ok_or(BuilderError::BadValue("incorrect vertex line format"))?
186                .parse()
187                .map_err(|_| BuilderError::BadValue("could not parse vertex ID"))?;
188            let x: f64 = it
189                .next()
190                .ok_or(BuilderError::BadValue("incorrect vertex line format"))?
191                .parse()
192                .map_err(|_| BuilderError::BadValue("could not parse vertex x coordinate"))?;
193            let y: f64 = it
194                .next()
195                .ok_or(BuilderError::BadValue("incorrect vertex line format"))?
196                .parse()
197                .map_err(|_| BuilderError::BadValue("could not parse vertex y coordinate"))?;
198            if it.next().is_some() {
199                return Err(BuilderError::BadValue("incorrect vertex line format"));
200            }
201            map.set_vertex(id, Vertex2(T::from(x).unwrap(), T::from(y).unwrap()));
202        }
203    }
204
205    Ok(map)
206}

322                    T::from(y).unwrap(),
323                    T::from(z).unwrap(),
324                ),

326        }
327    }

329    Ok(map)

334/// Node identifier used by an Abaqus input.
335type InpNodeId = usize;
336
337/// Parsed nodes and C3D8 element connectivity from an Abaqus INP document.
338struct InpFile {
339    /// Node coordinates indexed by their identifier.
340    nodes: HashMap<InpNodeId, [f64; 3]>,
341    /// C3D8 elements stored in Abaqus local node order.
342    elements: Vec<[InpNodeId; 8]>,
343}
344
345/// Section currently being read from an Abaqus INP document.
346enum InpSection {
347    /// Data outside a supported section.
348    None,
349    /// A `*NODE` section.
350    Nodes,
351    /// A supported `*ELEMENT` section.
352    Elements,
353}
354
355impl TryFrom<&str> for InpFile {
356    type Error = BuilderError;
357
358    /// Parse the supported subset of Abaqus INP data.
359    #[allow(clippy::too_many_lines)]
360    fn try_from(value: &str) -> Result<Self, Self::Error> {
361        let mut nodes = HashMap::default();
362        let mut elements = Vec::new();
363        let mut element_ids = HashSet::<usize>::default();
364        let mut section = InpSection::None;
365
366        for raw_line in value.lines() {
367            let line = raw_line.trim();
368            if line.is_empty() || line.starts_with("**") {
369                continue;
370            }
371
372            if line.starts_with('*') {
373                let mut header = line.split(',');
374                let keyword = header.next().expect("E: non-empty header").trim();
375                if keyword.eq_ignore_ascii_case("*NODE") {
376                    section = InpSection::Nodes;
377                } else if keyword.eq_ignore_ascii_case("*ELEMENT") {
378                    let element_type = header.find_map(|parameter| {
379                        let (name, value) = parameter.split_once('=')?;
380                        name.trim()
381                            .eq_ignore_ascii_case("TYPE")
382                            .then(|| value.trim().to_ascii_uppercase())
383                    });
384                    let Some(element_type) = element_type else {
385                        return Err(BuilderError::BadInpData(
386                            "element section has no TYPE parameter",
387                        ));
388                    };
389                    if !element_type.starts_with("C3D8") {
390                        return Err(BuilderError::UnsupportedInpData(
391                            "only C3D8 element types are supported",
392                        ));
393                    }
394                    section = InpSection::Elements;
395                } else {
396                    section = InpSection::None;
397                }
398                continue;
399            }
400
401            let fields = line.split(',').map(str::trim).collect::<Vec<_>>();
402            match section {
403                InpSection::None => {}
404                InpSection::Nodes => {
405                    if fields.len() != 4 || fields.iter().any(|field| field.is_empty()) {
406                        return Err(BuilderError::BadInpData("invalid node record"));
407                    }
408                    let node_id = fields[0]
409                        .parse()
410                        .map_err(|_| BuilderError::BadInpData("invalid node identifier"))?;
411                    let coordinates = [
412                        fields[1]
413                            .parse()
414                            .map_err(|_| BuilderError::BadInpData("invalid node coordinate"))?,
415                        fields[2]
416                            .parse()
417                            .map_err(|_| BuilderError::BadInpData("invalid node coordinate"))?,
418                        fields[3]
419                            .parse()
420                            .map_err(|_| BuilderError::BadInpData("invalid node coordinate"))?,
421                    ];
422                    if nodes.insert(node_id, coordinates).is_some() {
423                        return Err(BuilderError::BadInpData("duplicate node identifier"));
424                    }
425                }
426                InpSection::Elements => {
427                    if fields.len() != 9 || fields.iter().any(|field| field.is_empty()) {
428                        return Err(BuilderError::BadInpData("invalid C3D8 element record"));
429                    }
430                    let element_id = fields[0]
431                        .parse()
432                        .map_err(|_| BuilderError::BadInpData("invalid element identifier"))?;
433                    if !element_ids.insert(element_id) {
434                        return Err(BuilderError::BadInpData("duplicate element identifier"));
435                    }
436                    let mut element = [0; 8];
437                    for (node_id, field) in element.iter_mut().zip(fields.iter().skip(1)) {
438                        *node_id = field.parse().map_err(|_| {
439                            BuilderError::BadInpData("invalid element node identifier")
440                        })?;
441                    }
442                    if element.into_iter().collect::<HashSet<_>>().len() != 8 {
443                        return Err(BuilderError::BadInpData(
444                            "C3D8 element contains duplicate nodes",
445                        ));
446                    }
447                    elements.push(element);
448                }
449            }
450        }
451
452        if nodes.is_empty() {
453            return Err(BuilderError::BadInpData("file contains no nodes"));
454        }
455        if elements.is_empty() {
456            return Err(BuilderError::BadInpData("file contains no C3D8 elements"));
457        }
458        if elements
459            .iter()
460            .flatten()
461            .any(|node_id| !nodes.contains_key(node_id))
462        {
463            return Err(BuilderError::BadInpData(
464                "element references an undefined node",
465            ));
466        }
467
468        Ok(Self { nodes, elements })
469    }
470}

472/// Outward-oriented quadrilateral faces, expressed in Abaqus C3D8 local node order.
473const HEX_FACES: [[usize; 4]; 6] = [
474    [0, 1, 2, 3],
475    [1, 0, 4, 5],
476    [2, 1, 5, 6],
477    [3, 2, 6, 7],
478    [0, 3, 7, 4],
479    [5, 4, 7, 6],
480];
481
482/// First dart of each face in the internal 24-dart hexahedron representation.
483const HEX_FACE_DART_OFFSETS: [DartIdType; 6] = [0, 4, 8, 12, 16, 20];
484
485/// Representative dart of each vertex in the internal hexahedron representation.
486const HEX_VERTEX_DART_OFFSETS: [DartIdType; 8] = [0, 1, 2, 3, 6, 7, 11, 15];
487
488/// Beta-2 image of each dart, expressed as an offset within its hexahedron.
489const HEX_BETA_2_OFFSETS: [DartIdType; 24] = [
490    4, 8, 12, 16, 0, 19, 20, 9, 1, 7, 23, 13, 2, 11, 22, 17, 3, 15, 21, 5, 6, 18, 14, 10,
491];
492
493/// Initialize the beta relations of one 24-dart hexahedron.
494fn initialize_hex<T: CoordsFloat>(map: &CMap3<T>, first_dart: DartIdType) {
495    for face_offset in HEX_FACE_DART_OFFSETS {
496        for edge_offset in 0..4 {
497            let dart = first_dart + face_offset + edge_offset;
498            let previous = first_dart + face_offset + (edge_offset + 3) % 4;
499            let next = first_dart + face_offset + (edge_offset + 1) % 4;
500            let beta_2 = first_dart + HEX_BETA_2_OFFSETS[(face_offset + edge_offset) as usize];
501            map.set_betas(dart, [previous, next, beta_2, 0]);
502        }
503    }
504}

506/// Face waiting to be matched with an adjacent element.
507#[derive(Clone, Copy)]
508struct PendingFace {
509    /// Face nodes in their oriented local order.
510    nodes: [InpNodeId; 4],
511    /// First dart of the corresponding quadrilateral face.
512    first_dart: DartIdType,
513}
514
515/// Current sewing state of a face shared by mesh elements.
516enum FaceState {
517    /// The first occurrence of a face.
518    Pending(PendingFace),
519    /// A face already shared by two elements.
520    Sewn,
521}
522
523/// Return an orientation-independent key for a quadrilateral face.
524fn face_key(mut nodes: [InpNodeId; 4]) -> [InpNodeId; 4] {
525    nodes.sort_unstable();
526    nodes
527}

529/// Build a 3-map from the mesh contained in an Abaqus INP document.
530pub(crate) fn build_3d_from_inp<T: CoordsFloat>(
531    content: &str,
532    manager: AttrStorageManager,
533) -> Result<CMap3<T>, BuilderError> {
534    let input = InpFile::try_from(content)?;
535    let n_darts = input
536        .elements
537        .len()
538        .checked_mul(24)
539        .filter(|count| *count <= DartIdType::MAX as usize)
540        .ok_or(BuilderError::BadInpData("mesh contains too many elements"))?;
541    let map = CMap3::new_with_undefined_attributes(n_darts, manager);
542    let mut faces = HashMap::<[InpNodeId; 4], FaceState>::default();
543
544    for (element_index, element) in input.elements.iter().enumerate() {
545        let first_dart = 1 + DartIdType::try_from(element_index * 24)
546            .map_err(|_| BuilderError::BadInpData("mesh contains too many elements"))?;
547        initialize_hex(&map, first_dart);
548
549        for (face, &dart_offset) in HEX_FACES.iter().zip(&HEX_FACE_DART_OFFSETS) {
550            let ordered_nodes = face.map(|index| element[index]);
551            let key = face_key(ordered_nodes);
552            match faces.get_mut(&key) {
553                None => {
554                    faces.insert(
555                        key,
556                        FaceState::Pending(PendingFace {
557                            nodes: ordered_nodes,
558                            first_dart: first_dart + dart_offset,
559                        }),
560                    );
561                }
562                Some(state @ FaceState::Pending(_)) => {
563                    let FaceState::Pending(previous) = *state else {
564                        unreachable!()
565                    };
566                    let Some(reverse_edge_offset) = (0..4).find(|&index| {
567                        ordered_nodes[index] == previous.nodes[1]
568                            && ordered_nodes[(index + 1) % 4] == previous.nodes[0]
569                    }) else {
570                        return Err(BuilderError::BadInpData(
571                            "adjacent elements have inconsistent face orientations",
572                        ));
573                    };
574                    map.link::<3>(
575                        previous.first_dart,
576                        first_dart + dart_offset + reverse_edge_offset as DartIdType,
577                    )
578                    .map_err(|_| BuilderError::BadInpData("could not link adjacent elements"))?;
579                    *state = FaceState::Sewn;
580                }
581                Some(FaceState::Sewn) => {
582                    return Err(BuilderError::BadInpData(
583                        "face is shared by more than two elements",
584                    ));
585                }
586            }
587        }
588    }
589
590    // A node ID normally maps to one vertex orbit. If a deck reuses an ID across topologically
591    // disconnected components, each orbit still needs its own copy of the coordinate.
592    let mut vertex_nodes = HashMap::<VertexIdType, InpNodeId>::default();
593    for (element_index, element) in input.elements.iter().enumerate() {
594        let first_dart = 1 + element_index as DartIdType * 24;
595        for (&node_id, &dart_offset) in element.iter().zip(&HEX_VERTEX_DART_OFFSETS) {
596            let vertex_id = map.vertex_id(first_dart + dart_offset);
597            if vertex_nodes
598                .insert(vertex_id, node_id)
599                .is_some_and(|previous| previous != node_id)
600            {
601                return Err(BuilderError::BadInpData(
602                    "topology merges different node identifiers",
603                ));
604            }
605        }
606    }
607
608    for (vertex_id, node_id) in vertex_nodes {
609        let [x, y, z] = input.nodes[&node_id];
610        let vertex = Vertex3(
611            T::from(x).ok_or(BuilderError::BadInpData(
612                "cannot represent coordinate value in desired FP type",
613            ))?,
614            T::from(y).ok_or(BuilderError::BadInpData(
615                "cannot represent coordinate value in desired FP type",
616            ))?,
617            T::from(z).ok_or(BuilderError::BadInpData(
618                "cannot represent coordinate value in desired FP type",
619            ))?,
620        );
621        map.set_vertex(vertex_id, vertex);
622    }
623
624    Ok(map)
625}

629macro_rules! if_predicate_return_err {
630    ($pr: expr, $er: expr) => {
631        if $pr {
632            return Err($er);
633        }
634    };

637macro_rules! build_vertices {
638    ($v: ident) => {{
639        if_predicate_return_err!(
640            !($v.len() % 3).is_zero(),
641            BuilderError::BadVtkData("vertex list contains an incomplete tuple")
642        );
643        $v.chunks_exact(3)
644            .map(|slice| {
645                // WE IGNORE Z values
646                let &[x, y, _] = slice else { unreachable!() };
647                Vertex2(T::from(x).unwrap(), T::from(y).unwrap())
648            })
649            .collect()
650    }};
651}

653// ------ building routine
654
655#[allow(clippy::too_many_lines)]
656/// Internal building routine for [`CMap2::from_vtk_file`].
657///
658/// # Result / Errors
659///
660/// This implementation support only a very specific subset of VTK files. This result in many
661/// possibilities for failure. This function may return:
662///
663/// - `Ok(CMap2)` -- The file was successfully parsed and its content made into a 2-map.
664/// - `Err(BuilderError)` -- The function failed for one of the following reasons (sorted
665///   by [`BuilderError`] variants):
666///     - `UnsupportedVtkData`: The file contains unsupported data, i.e.:
667///         - file format isn't Legacy,
668///         - data set is something other than `UnstructuredGrid`,
669///         - coordinate representation type isn't `float` or `double`
670///         - mesh contains unsupported cell types (`PolyVertex`, `PolyLine`, `TriangleStrip`,
671///           `Pixel` or anything 3D)
672///     - `InvalidVtkFile`: The file contains inconsistencies, i.e.:
673///         - the number of coordinates cannot be divided by `3`, meaning a tuple is incomplete
674///         - the number of `Cells` and `CellTypes` isn't equal
675///         - a given cell has an inconsistent number of vertices with its specified cell type
676pub fn build_2d_from_vtk<T: CoordsFloat>(
677    value: Vtk,
678    mut _manager: AttrStorageManager, // FIXME: find a cleaner solution to populate the manager
679) -> Result<CMap2<T>, BuilderError> {
680    let mut cmap: CMap2<T> = CMap2::new(0);
681    let mut sew_buffer: BTreeMap<(usize, usize), DartIdType> = BTreeMap::new();
682    match value.data {
683        DataSet::ImageData { .. }
684        | DataSet::StructuredGrid { .. }
685        | DataSet::RectilinearGrid { .. }
686        | DataSet::PolyData { .. }
687        | DataSet::Field { .. } => {
688            return Err(BuilderError::UnsupportedVtkData("dataset not supported"));
689        }
690        DataSet::UnstructuredGrid { pieces, .. } => {
691            let mut tmp = pieces.iter().map(|piece| {
692                // assume inline data
693                let Ok(tmp) = piece.load_piece_data(None) else {
694                    return Err(BuilderError::UnsupportedVtkData("not inlined data piece"));
695                };
696
697                // build vertex list
698                // since we're expecting coordinates, we'll assume floating type
699                // we're also converting directly to our vertex type since we're building a 2-map
700                let vertices: Vec<Vertex2<T>> = match tmp.points {
701                    IOBuffer::F64(v) => build_vertices!(v),
702                    IOBuffer::F32(v) => build_vertices!(v),
703                    _ => {
704                        return Err(BuilderError::UnsupportedVtkData(
705                            "unsupported coordinate type",
706                        ));
707                    }
708                };
709
710                let vtkio::model::Cells { cell_verts, types } = tmp.cells;
711                match cell_verts {
712                    VertexNumbers::Legacy {
713                        num_cells,
714                        vertices: verts,
715                    } => {
716                        // check basic stuff
717                        if_predicate_return_err!(
718                            num_cells as usize != types.len(),
719                            BuilderError::BadVtkData("different # of cell in CELLS and CELL_TYPES")
720                        );
721
722                        // build a collection of vertex lists corresponding of each cell
723                        let mut cell_components: Vec<Vec<usize>> = Vec::new();
724                        let mut take_next = 0;
725                        for vertex_id in &verts {
726                            if take_next.is_zero() {
727                                // making it usize since it's a counter
728                                take_next = *vertex_id as usize;
729                                cell_components.push(Vec::with_capacity(take_next));
730                            } else {
731                                cell_components
732                                    .last_mut()
733                                    .expect("E: unreachable")
734                                    .push(*vertex_id as usize);
735                                take_next -= 1;
736                            }
737                        }
738                        assert_eq!(num_cells as usize, cell_components.len());
739
740                        let mut errs =
741                            types
742                                .iter()
743                                .zip(cell_components.iter())
744                                .map(|(cell_type, vids)| match cell_type {
745                                    CellType::Vertex => {
746                                        if_predicate_return_err!(
747                                            vids.len() != 1,
748                                            BuilderError::BadVtkData(
749                                                "`Vertex` with incorrect # of vertices (!=1)"
750                                            )
751                                        );
752                                        // silent ignore
753                                        Ok(())
754                                    }
755                                    CellType::PolyVertex => Err(BuilderError::UnsupportedVtkData(
756                                        "`PolyVertex` cell type",
757                                    )),
758                                    CellType::Line => {
759                                        if_predicate_return_err!(
760                                            vids.len() != 2,
761                                            BuilderError::BadVtkData(
762                                                "`Line` with incorrect # of vertices (!=2)"
763                                            )
764                                        );
765                                        // silent ignore
766                                        Ok(())
767                                    }
768                                    CellType::PolyLine => Err(BuilderError::UnsupportedVtkData(
769                                        "`PolyLine` cell type",
770                                    )),
771                                    CellType::Triangle => {
772                                        // check validity
773                                        if_predicate_return_err!(
774                                            vids.len() != 3,
775                                            BuilderError::BadVtkData(
776                                                "`Triangle` with incorrect # of vertices (!=3)"
777                                            )
778                                        );
779                                        // build the triangle
780                                        let d0 = cmap.allocate_used_darts(3);
781                                        let (d1, d2) = (d0 + 1, d0 + 2);
782                                        cmap.set_vertex(d0 as VertexIdType, vertices[vids[0]]);
783                                        cmap.set_vertex(d1 as VertexIdType, vertices[vids[1]]);
784                                        cmap.set_vertex(d2 as VertexIdType, vertices[vids[2]]);
785                                        cmap.link::<1>(d0, d1).unwrap(); // edge d0 links vertices vids[0] & vids[1]
786                                        cmap.link::<1>(d1, d2).unwrap(); // edge d1 links vertices vids[1] & vids[2]
787                                        cmap.link::<1>(d2, d0).unwrap(); // edge d2 links vertices vids[2] & vids[0]
788                                        // record a trace of the built cell for future 2-sew
789                                        sew_buffer.insert((vids[0], vids[1]), d0);
790                                        sew_buffer.insert((vids[1], vids[2]), d1);
791                                        sew_buffer.insert((vids[2], vids[0]), d2);
792                                        Ok(())
793                                    }
794                                    CellType::TriangleStrip => {
795                                        Err(BuilderError::UnsupportedVtkData(
796                                            "`TriangleStrip` cell type",
797                                        ))
798                                    }
799                                    CellType::Polygon => {
800                                        let n_vertices = vids.len();
801                                        let d0 = cmap.allocate_used_darts(n_vertices);
802                                        (0..n_vertices).for_each(|i| {
803                                            let di = d0 + i as DartIdType;
804                                            let dip1 =
805                                                if i == n_vertices - 1 { d0 } else { di + 1 };
806                                            cmap.set_vertex(di as VertexIdType, vertices[vids[i]]);
807                                            cmap.link::<1>(di, dip1).unwrap();
808                                            sew_buffer
809                                                .insert((vids[i], vids[(i + 1) % n_vertices]), di);
810                                        });
811                                        Ok(())
812                                    }
813                                    CellType::Pixel => {
814                                        Err(BuilderError::UnsupportedVtkData("`Pixel` cell type"))
815                                    }
816                                    CellType::Quad => {
817                                        if_predicate_return_err!(
818                                            vids.len() != 4,
819                                            BuilderError::BadVtkData(
820                                                "`Quad` with incorrect # of vertices (!=4)"
821                                            )
822                                        );
823                                        // build the quad
824                                        let d0 = cmap.allocate_used_darts(4);
825                                        let (d1, d2, d3) = (d0 + 1, d0 + 2, d0 + 3);
826                                        cmap.set_vertex(d0 as VertexIdType, vertices[vids[0]]);
827                                        cmap.set_vertex(d1 as VertexIdType, vertices[vids[1]]);
828                                        cmap.set_vertex(d2 as VertexIdType, vertices[vids[2]]);
829                                        cmap.set_vertex(d3 as VertexIdType, vertices[vids[3]]);
830                                        cmap.link::<1>(d0, d1).unwrap(); // edge d0 links vertices vids[0] & vids[1]
831                                        cmap.link::<1>(d1, d2).unwrap(); // edge d1 links vertices vids[1] & vids[2]
832                                        cmap.link::<1>(d2, d3).unwrap(); // edge d2 links vertices vids[2] & vids[3]
833                                        cmap.link::<1>(d3, d0).unwrap(); // edge d3 links vertices vids[3] & vids[0]
834                                        // record a trace of the built cell for future 2-sew
835                                        sew_buffer.insert((vids[0], vids[1]), d0);
836                                        sew_buffer.insert((vids[1], vids[2]), d1);
837                                        sew_buffer.insert((vids[2], vids[3]), d2);
838                                        sew_buffer.insert((vids[3], vids[0]), d3);
839                                        Ok(())
840                                    }
841                                    _ => Err(BuilderError::UnsupportedVtkData(
842                                        "CellType not supported in 2-maps",
843                                    )),
844                                });
845                        if let Some(is_err) = errs.find(Result::is_err) {
846                            return Err(is_err.unwrap_err()); // unwrap & wrap because type inference is clunky
847                        }
848                    }
849                    VertexNumbers::XML { .. } => {
850                        return Err(BuilderError::UnsupportedVtkData("XML format"));
851                    }
852                }
853                Ok(())
854            });
855            // return the first error if there is one
856            if let Some(is_err) = tmp.find(Result::is_err) {
857                return Err(is_err.unwrap_err()); // unwrap & wrap because type inference is clunky
858            }
859        }
860    }
861    while let Some(((id0, id1), dart_id0)) = sew_buffer.pop_first() {
862        if let Some(dart_id1) = sew_buffer.remove(&(id1, id0)) {
863            cmap.sew::<2>(dart_id0, dart_id1).unwrap();
864        }
865    }
866    Ok(cmap)
867}
